git config --global user.name "Your Name"
-> Sets the global Git username.

git config --global user.email "email@example.com"
-> Sets the global Git email.

mkdir git-assignment
-> Creates a new directory.

cd git-assignment
-> Enters the assignment directory.

echo "<h1>Home</h1>" > index.html
-> Creates index.html with HTML content.

echo "<h1>About</h1>" > about.html
-> Creates about.html with HTML content.

git init -b main
-> Initializes a Git repository with main as the default branch.

ls
-> Lists files and directories.

git status
-> Shows staged, unstaged and untracked files.

git add .
-> Stages all changes for the next commit.

git add <filename>
-> Stages a specific file.

git commit -m "message"
-> Saves staged changes as a commit with a message.

git branch feature
-> Creates a new branch named feature.

git branch
-> Lists all local branches.

git switch <name>
-> Switches to the given branch.

git merge feature --no-ff -m "message"
-> Merges feature into the current branch and creates a merge commit.

git log --oneline
-> Shows compact commit history.

git log --oneline --graph --all
-> Shows commit history with a graph of all branches.

git reset --hard HEAD~1
-> Removes the last commit and discards its changes.

echo "*.log" > .gitignore
-> Adds a rule to ignore all .log files.

cat .gitignore
-> Displays the contents of .gitignore.

git remote add origin <url>
-> Links the local repository to a GitHub repository.

git remote -v
-> Lists configured remote repositories.

git push -u origin main
-> Pushes main to GitHub and sets its upstream branch.

git push origin feature
-> Pushes the feature branch to GitHub.
